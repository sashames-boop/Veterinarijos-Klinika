    -username: String
    -pass: String

class User {
    -id: Integer
    -name: String
    -username: String
    -phone: String
    -password: String
    -blocked: Boolean
    +isBlocked(): boolean
    +setBlocked(blocked: boolean): void
}

class Client {
    -email: String
    +registerPet(pet: Pet): void
    +bookAppointment(date: String, time: String, pet: Pet, vet: Vet): Registration
}

class Pet {
    -id: Integer
    -name: String
    -type: PetType
    -age: int
    -weight: double
    +getInfo(): String
}

class Vet {
    -specialization: String
    +acceptPatient(registration: Registration): void
    +confirmRegistration(): void
}

class Groomer {
    -specialization: String
    +acceptPatient(registration: Registration): void
    +confirmRegistration(): void
}

class Registration {
    -id: Integer
    -date: String
    -time: String
    -confirmed: boolean
    -cancelled: Boolean
    +confirm(): void
    +cancel(): void
    +setDate(date: String): void
    +setTime(time: String): void
}

class Incident {
    -id: int
    -registrationDate: LocalDateTime
    -modificationDate: LocalDateTime
    -status: IncidentStatus
    -comment: String
}

enum PetType {
    DOG
    CAT
    RABBIT
    FISH
    MOUSE
    BIRD
    REPTILE
    HAMSTER
    GUINEA
    OTHER
}

enum IncidentStatus {
    CREATED
    ACCEPTED
    RESOLVED
    CANCELLED
}

class Clinic {

    -address: String
    -free_spots: int

    +redaguoti_pasl(): void //+ keisti kaina; kokiems gyvunams teikiama paslauga

    +nustatyti_darbo_laika(): LocalTime
}
